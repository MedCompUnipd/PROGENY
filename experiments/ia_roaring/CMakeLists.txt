cmake_minimum_required(VERSION 3.20)
project(progeny_ia_roaring_experiment LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenMP REQUIRED)
file(GLOB_RECURSE CROARING_SOURCES CONFIGURE_DEPENDS vendor/CRoaring/src/*.c)
set(CMAKE_C_STANDARD 11)
add_library(progeny_croaring STATIC ${CROARING_SOURCES})
target_include_directories(progeny_croaring PUBLIC vendor/CRoaring/include)
target_compile_definitions(progeny_croaring PUBLIC CROARING_COMPILER_SUPPORTS_AVX512=0)
add_executable(progeny-ia-roaring-v1 src/main.cpp)
target_link_libraries(progeny-ia-roaring-v1 PRIVATE progeny_croaring OpenMP::OpenMP_CXX)
